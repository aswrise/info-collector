§1

数据模型

已定 · 按此建库

单文件 SQLite（~/.info-collector/source-sync/source-sync.db，标准库 sqlite3），6 张业务表 + 1 张元数据表。全部使用 STRICT 表和 CHECK 枚举约束，时间戳统一存 ISO-8601 UTC 文本。以下 DDL 即最终规格。

连接初始化（每个进程建连后执行）

PRAGMA journal_mode = WAL;    -- 读写并发：Dashboard 查询不被扫描/Worker 写入阻塞
PRAGMA synchronous = NORMAL;  -- WAL 模式下安全且明显更快
PRAGMA foreign_keys = ON;
PRAGMA busy_timeout = 5000;

meta — 版本与迁移（可拓展性的地基）

CREATE TABLE meta (
  key   TEXT PRIMARY KEY,
  value TEXT NOT NULL
) STRICT;
-- schema_version 从 1 开始。之后所有结构变更写成顺序迁移脚本
-- migrations/0002_*.sql …，启动时按版本号依次应用，禁止手改线上库。
决策 1.1 — 全局内容主键 content_key已定

单列文本主键，格式 youtube:<video_id> / x:<tweet_id>。可读、可 grep、可直接出现在日志里；前缀方案天然可拓展——未来接 RSS 或其他平台时新增 rss:<hash> 之类前缀即可，不动表结构。

sources — 监控来源

CREATE TABLE sources (
  id                INTEGER PRIMARY KEY,
  source_type       TEXT NOT NULL CHECK (source_type IN
                      ('youtube_channel','youtube_playlist','x_likes','x_list')),
  canonical_url     TEXT NOT NULL UNIQUE, -- 规范化 URL，防重复添加
  external_id       TEXT NOT NULL,        -- channel_id / playlist_id / handle / list_id
  display_name      TEXT NOT NULL,
  enabled           INTEGER NOT NULL DEFAULT 1 CHECK (enabled IN (0,1)),
  auto_sync_new     INTEGER NOT NULL DEFAULT 0 CHECK (auto_sync_new IN (0,1)),
  collection_policy TEXT NOT NULL CHECK (collection_policy IN
                      ('auto_sync_new','always_collect','value_filter')),
  settings_json     TEXT NOT NULL DEFAULT '{}',
  last_scan_at      TEXT,
  last_scan_status  TEXT CHECK (last_scan_status IN
                      ('complete','incomplete','failed','paused_risk') OR last_scan_status IS NULL),
  last_scan_error   TEXT,
  created_at        TEXT NOT NULL
) STRICT;
决策 1.2 — settings_json 内容已定
  • YouTube Channel：{"scan_depth": 5}
  • X Likes：{"page_interval_seconds": [5, 10], "backfill_pages_per_run": 15}
  • X List：{"topics": […], "minimum_collect_score": 7, "auto_sync_accepted": false, "send_borderline_to_review": true, "shadow_mode": true}

JSON 列只在扫描时读，不参与 Dashboard 筛选，因此不需要可索引。新来源类型的专属设置继续进这个列，不加新列。

content_items — 全局内容实体

CREATE TABLE content_items (
  content_key    TEXT PRIMARY KEY,
  platform       TEXT NOT NULL CHECK (platform IN ('youtube','x')),
  canonical_url  TEXT NOT NULL,
  title_or_text  TEXT,
  author         TEXT,
  published_at   TEXT,            -- 原帖/视频发布时间，不是发现时间
  language       TEXT CHECK (language IN ('zh','en','mixed','unknown') OR language IS NULL),
  metadata_json  TEXT NOT NULL DEFAULT '{}',  -- 引用 Tweet、外链卡片、媒体、作者简介等
  created_at     TEXT NOT NULL
) STRICT;

CREATE INDEX idx_items_platform ON content_items(platform, published_at);

source_relationships — 来源关系

CREATE TABLE source_relationships (
  source_id     INTEGER NOT NULL REFERENCES sources(id),
  content_key   TEXT NOT NULL REFERENCES content_items(content_key),
  status        TEXT NOT NULL DEFAULT 'present'
                  CHECK (status IN ('present','removed','unavailable')),
  first_seen_at TEXT NOT NULL,  -- 首次观察时间；Likes 场景 ≈ 点赞时间，但不冒充它
  last_seen_at  TEXT NOT NULL,
  position      INTEGER,        -- Playlist 顺序；其余来源为 NULL
  PRIMARY KEY (source_id, content_key)
) STRICT;

-- 反查内容出现在哪些来源：Likes 优先级规则、详情抽屉都靠它
CREATE INDEX idx_rel_content ON source_relationships(content_key);

sync_jobs — 同步任务

CREATE TABLE sync_jobs (
  id           INTEGER PRIMARY KEY,
  content_key  TEXT NOT NULL REFERENCES content_items(content_key),
  processor    TEXT NOT NULL CHECK (processor IN ('podcast','tweet_organizer')),
  status       TEXT NOT NULL CHECK (status IN ('queued','syncing','synced','failed')),
  trigger      TEXT NOT NULL CHECK (trigger IN ('auto','user','initial_import')),
  attempts     INTEGER NOT NULL DEFAULT 0,
  last_error   TEXT,
  result_json  TEXT,            -- 产物路径、translation_status 等
  created_at   TEXT NOT NULL,
  started_at   TEXT,
  finished_at  TEXT
) STRICT;

-- 幂等核心：同一 (content_key, processor) 的活跃任务全库唯一。
-- failed 不在部分索引内，因此可重新入队。
CREATE UNIQUE INDEX uq_active_job
  ON sync_jobs(content_key, processor)
  WHERE status IN ('queued','syncing','synced');

-- Worker 认领 + Dashboard 状态计数都走这条索引
CREATE INDEX idx_jobs_status ON sync_jobs(status, created_at);
决策 1.3 — 入队与认领的并发语义已定

入队用 INSERT … ON CONFLICT DO NOTHING 打在部分唯一索引上，按受影响行数统计 {queued, skipped_synced, skipped_active}——无竞态、单条 SQL。Worker 认领用 UPDATE … SET status='syncing', started_at=… WHERE id IN (SELECT … WHERE status='queued' ORDER BY created_at LIMIT ?) RETURNING *，单事务完成。syncing 超过 30 分钟视为超时，由下一次 Worker 启动时恢复为 queued（attempts 不清零）。

value_decisions — X List 价值判定

CREATE TABLE value_decisions (
  content_key      TEXT NOT NULL REFERENCES content_items(content_key),
  source_id        INTEGER NOT NULL REFERENCES sources(id),
  relevance        INTEGER NOT NULL CHECK (relevance BETWEEN 0 AND 3),
  information_gain INTEGER NOT NULL CHECK (information_gain BETWEEN 0 AND 3),
  usefulness       INTEGER NOT NULL CHECK (usefulness BETWEEN 0 AND 2),
  evidence         INTEGER NOT NULL CHECK (evidence BETWEEN 0 AND 2),
  total            INTEGER NOT NULL CHECK (total BETWEEN 0 AND 10),
  decision         TEXT NOT NULL CHECK (decision IN ('collect','review','noise')),
  reason_code      TEXT CHECK (reason_code IN ('social_chatter','contextless_reaction',
                     'pure_promotion','engagement_bait','repetition','off_topic',
                     'insufficient_context') OR reason_code IS NULL),
  explanation      TEXT,
  confidence       REAL,
  overridden_by    TEXT CHECK (overridden_by IN ('user_collect','user_noise') OR overridden_by IS NULL),
  overridden_at    TEXT,
  decided_at       TEXT NOT NULL,
  PRIMARY KEY (content_key, source_id)
) STRICT;

-- review 队列与影子模式统计
CREATE INDEX idx_decisions_review ON value_decisions(source_id, decision);
决策 1.4 — 判定按 (内容 × 来源) 存，覆盖不改原始分数已定

同一条 Tweet 在不同 List 各自判定。用户覆盖只写 overridden_by，四项原始分数保留用于日后校准（第一版只存纠正记录，不训练模型）。「Likes 优先」不在这张表上做：入队逻辑发现 Likes 来源关系存在时直接绕过 value_decisions。

source_checkpoints — 扫描断点与水位

CREATE TABLE source_checkpoints (
  source_id            INTEGER PRIMARY KEY REFERENCES sources(id),
  overlap_ids_json     TEXT,  -- X：上次完整扫描末尾 5 个已落库 tweet_id（水位）
  pending_cursor       TEXT,  -- X：中断/分批回溯时保存的 next_cursor，恢复前须验证有效
  pending_stop_reason  TEXT CHECK (pending_stop_reason IN
                         ('rate_limited','auth','challenge','error','page_budget')
                         OR pending_stop_reason IS NULL),
  snapshot_json        TEXT,  -- YouTube Playlist：上次完整快照（video_id 有序列表）
  last_complete_at     TEXT,
  last_complete_pages  INTEGER
) STRICT;
决策 1.5 — 水位 = 5 个 overlap ID，命中即停已定

停止条件是「当前页出现任意一个 overlap_id」，判断的是已发现落库而非已整理完成——发现和处理解耦。5 个而不是 1 个，防止取消点赞恰好删掉水位帖。异常或页数预算（page_budget，见 §3）中止时：已落库页面保留、水位不前移、pending_cursor 存下来，Dashboard 显示 incomplete。

性能说明：本系统数据量级为数千行，以上索引已覆盖全部高频查询（Worker 认领、状态计数、内容反查、review 队列）。刻意不做的：不加 FTS 全文索引（关键词筛选用 LIKE 足够，量级大了再迁移）、不分库、不加缓存层。

§2

新模块 Interface

已定 · 按此实现

五个新 Interface，全部是纯 Python（dataclass + Protocol），无 Web 框架。SQLite 表、事务、状态迁移隐藏在 Source Registry 内部，Adapter 和 Processor 都不直接碰数据库。

共享类型（source_sync/types.py）

@dataclass(frozen=True)
class DiscoveredItem:            # Adapter 产出的最小单位
    content_key: str              # "youtube:abc123" / "x:189…"
    canonical_url: str
    title_or_text: str
    author: str
    published_at: str | None
    position: int | None          # 来源内顺序（Playlist 用）
    metadata: dict                # 平台原始数据，X 含引用/媒体/外链卡片

@dataclass(frozen=True)
class ScanPage:                  # 一页扫描结果（X）或完整快照（YouTube）
    items: list[DiscoveredItem]
    next_checkpoint: str | None   # X cursor；None = 无下一页
    exhausted: bool               # 已到来源末尾
    is_snapshot: bool             # True = Playlist 全量快照，做 diff 而非追加

@dataclass(frozen=True)
class SourcePreview:             # inspect() 返回，供添加来源确认页展示
    source_type: str
    canonical_url: str
    external_id: str
    display_name: str
    item_count: int | None
    recent_items: list[DiscoveredItem]

Source Adapter（每种来源一个实现）

class SourceAdapter(Protocol):
    def inspect(self, url: str) -> SourcePreview: ...
    def scan(self, source: Source, checkpoint: str | None) -> ScanPage: ...

四个实现：YouTubeChannelAdapter / YouTubePlaylistAdapter（yt-dlp，返回快照）、XLikesAdapter / XListAdapter（OpenCLI，严格 20 条一页，返回带 cursor 的分页）。X 两个 adapter 共享分页与风控代码，endpoint 和解析器分开。

Source Registry（唯一持有 SQLite 的模块）

class SourceRegistry:
    def add_source(self, url, initial_policy) -> Source
    def record_scan(self, source_id, page: ScanPage) -> DiscoveryResult
        # 单页事务落库：内容 + 关系 + （按策略）入队，返回新增/水位命中情况
    def enqueue(self, content_keys, processor, trigger) -> EnqueueResult
        # EnqueueResult = {queued: int, skipped_synced: int, skipped_active: int}
    def claim_jobs(self, limit) -> list[Job]     # queued → syncing，事务内
    def finish_job(self, job_id, result) -> str  # → synced | failed

class PodcastProcessor:
    def lookup(self, canonical_youtube_url) -> Lookup
        # not_synced | synced（synced 时带三个产物路径）——添加来源时用它标已同步
    def process(self, canonical_youtube_url, metadata) -> ProcessResult

决策 2.1 — 幂等以「产物文件」为准，不以数据库为准已定

手动收藏系统和自动系统各有一套队列状态，两边不共享数据库。同一视频不重复生成的保证只能放在共享的 Processor 上：process() 开头检查 workdir 里的 result.json 和三个产物文件，齐全即返回。新系统的 synced 定义 = 文件齐全。

Tweet 整理稿文件规格

文件名 中文短标题（@handle）.md，重名时追加 -<tweet_id 后 6 位>。frontmatter 对齐现有 translate-article 惯例（title / source / published / date / authors），追加 Tweet 专属字段：

---
title: <中文短标题，由 Tweet Organizer 生成>
source: https://x.com/<handle>/status/<tweet_id>
published: <原帖发布日期 YYYY-MM-DD>
date: <收录时间 YYYY-MM-DDTHH:mm，必须取自系统时间，禁止凭记忆填写>
authors: <作者真实姓名（@handle）>
platform: x
tweet_id: "<tweet_id>"
language: zh | en | mixed
translation_status: not_needed | translated | partially_translated
origin: likes | list
lists: [<来源 List 名称，Likes 来源则为 Likes>]
tags: [<必要标签>]
---

正文规则照旧：中文整理不翻译；英文保留原文 + 中文翻译；混合只翻英文部分；保留引用、外链卡片和媒体 URL；不扩写。

Obsidian Base 规格（tweet 整理.base）

  • 过滤：file.folder 为 文章收集/tweet 整理
  • 默认视图「全部」：表格，按 date 倒序，列 = title / authors / published / language / translation_status / origin / tags
  • 视图「已翻译」：translation_status != not_needed
  • 视图「来自 List」：origin == list，多一列 lists

用户可见行为（自设计基线继承，未变）

3.1 首次同步选项（添加 YouTube 来源时）已定
  • 只建立监控，不同步历史
  • 同步最新 5 条（默认）
  • 同步最新 N 条 / 同步全部

措辞约束：Playlist 没有可靠加入时间，界面必须写「当前列表前 N 条」，不能写「最近加入 N 条」。X Likes 没有条数选项——逐页追到历史重合或末尾才算完成；X List 可设初始范围，默认进影子判定模式。

3.2 批量操作的反馈文案已定
已加入 4 条 · 跳过已同步 2 条 · 跳过排队中/处理中 1 条

三个数字直接来自 enqueue() 的返回值（决策 1.3），界面不自己数。

3.3 X List 影子模式默认开已定

新 List 接入后 Value Evaluator 只打分不入队。人工检查约 50–100 条、可导出样本校准后，手动开启 auto_sync_accepted。insufficient_context 必须落 review，noise 保留可见、可人工覆盖。同一 Tweet 只要出现在 Likes 就必须收集，即使在 List 中被判为 noise。

3.4 删除来源 ≠ 删除内容已定

「删除监控关系」只删来源和关系记录，内容与产物一律保留；来源里被移除的条目只把关系置为 removed。Dashboard 不提供删除产物的入口。

分阶段交付

阶段界面交付
1骨架：标题行 + 简版健康卡（仅 YouTube Channel）+ 筛选条 + 内容表 + 批量操作条 + 添加来源向导（inspect 预览 → 策略确认 → 首次同步范围）。抽屉先只承载「产物路径 / 失败原因」两种内容
2健康卡增加 Playlist 形态（快照 diff 摘要、removed/unavailable 计数）
3X Likes 健康卡（页数、水位是否接上、停止原因、回溯进度「已读 n 页 / cursor 待续」）；扫描进度轮询展示
4抽屉的价值判定形态 + review 批量 collect/noise + 影子模式校验计数（n/50）+ 影子样本导出

交互基调

  • 局部刷新（fetch + DOM 更新），不整页跳转；扫描/处理中状态 2 秒轮询 status.json
  • 状态一律用色 pill 编码：未同步 排队中 处理中 已同步 失败 review
